cmake_minimum_required(VERSION 3.16)
project(ly_power_control VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

option(LP_BUILD_EXAMPLES "Build the host demonstration" ON)
option(LP_BUILD_TESTS "Build the numerical regression tests" ON)
option(LP_BUILD_SIMULATIONS "Build the dynamic motor and energy simulation" ON)

add_library(ly_power STATIC src/power_control.c)
target_include_directories(ly_power PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
set_target_properties(ly_power PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
if(NOT MSVC)
    target_compile_options(ly_power PRIVATE -Wall -Wextra -Wpedantic)
    target_link_libraries(ly_power PUBLIC m)
endif()

if(LP_BUILD_EXAMPLES)
    add_executable(power_control_demo examples/power_control_demo.c)
    target_link_libraries(power_control_demo PRIVATE ly_power)
endif()

if(LP_BUILD_TESTS)
    enable_testing()
    add_executable(test_power_control tests/test_power_control.c)
    target_link_libraries(test_power_control PRIVATE ly_power)
    add_test(NAME power_control COMMAND test_power_control)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME model_fitting COMMAND ${Python3_EXECUTABLE}
            ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_fit_power_model.py)
    endif()
endif()

if(LP_BUILD_SIMULATIONS)
    add_executable(power_control_sim simulations/power_control_sim.c)
    target_link_libraries(power_control_sim PRIVATE ly_power)
    if(LP_BUILD_TESTS)
        add_test(NAME dynamic_simulation COMMAND power_control_sim --check-only)
    endif()
endif()
